# What a public commit message, patch message or pull request must not
# carry: one extended regex per line, matched case-insensitively by
# airdress/scripts/commit-messages.sh. Each says why above it.

# Internal document numbers: specs, research and decision records,
# runbooks, recipes, business and cost records.
(^|[^[:alnum:]])(spec|rdr|rcp|rb|brr|cts)[-_ ]?[0-9]{3}([^[:alnum:]]|$)

# Task ids of those documents ("185-A.4", "185-0.2") and the trailer that
# ticks them.
(^|[^[:alnum:]])[0-9]{3}-[a-z0-9]{1,3}\.[0-9]+[a-z]?([^[:alnum:]]|$)
^[[:space:]]*tasks-done:

# The internal tracker's issue keys.
(^|[^[:alnum:]])air-[0-9]+([^[:alnum:]]|$)

# Private repositories and the paths of the internal documents.
(^|[^[:alnum:]])airdress-(ops|specs|board)([^[:alnum:]]|$)
\.spec-workflow
(^|/)runbooks/
